# 天机阁 · 工程化架构设计 v1.0

> **编制日期**：2026-08-02
> **定位**：从个人项目 → 工程级领域知识推理系统
> **核心原则**：不删除已有术数计算能力、不简化规则、不修改核心计算结果、不替换为外部API

---

## 一、现状诊断摘要

| 维度 | 现状 | 问题 |
|------|------|------|
| 架构 | Python后端 + JS前端，双引擎实现 | 前后端算法重复，维护成本翻倍 |
| API | 单文件 `bazi_api.py` 530行，13端点 | 耦合度高，无版本管理 |
| 引擎 | 5个Python类 + 10个JS函数，返回格式各异 | 无统一接口，合参需专门适配 |
| 规则 | 散落在函数内部，硬编码阈值 | 不可配置，修改需改代码 |
| 知识 | Markdown文档 + 代码常量，无结构化 | 查询困难，无法被程序消费 |
| 大师 | 34位模板在 `masters-engine.js` 中 | 风格模板化，非策略驱动 |
| 测试 | `test_all.py` 冒烟 + `test_exam.py` 考试 | 无pytest，无CI |
| 部署 | 手动 `uvicorn` 启动 | 无Docker，无环境管理 |

## 三、核心模块设计

### 3.1 引擎标准化接口（`core/interfaces.py`）

```python
from abc import ABC, abstractmethod
from typing import Any, Dict, Optional
from pydantic import BaseModel
from datetime import datetime


class EngineInput(BaseModel):
    """引擎统一输入"""
    params: Dict[str, Any]          # 引擎特定参数
    context: Optional[Dict] = None  # 上下文（如用户问题、领域）
    trace_id: Optional[str] = None  # 推理链追踪ID


class EngineOutput(BaseModel):
    """引擎统一输出"""
    engine_name: str                # 引擎名称
    input_hash: str                 # 输入哈希（版本追踪）
    timestamp: datetime             # 计算时间

    # 三阶段输出
    calculation: Dict[str, Any]     # calculate() 计算结果
    analysis: Optional[Dict] = None # analyze() 分析结果
    explanation: Optional[str] = None # explain() 解释文本

    # 元数据
    version: str = "1.0"            # 引擎版本
    trace: Optional[Dict] = None    # 推理链
    warnings: list = []             # 警告信息


class BaseEngine(ABC):
    """所有计算引擎的抽象基类"""

    @property
    @abstractmethod
    def name(self) -> str:
        """引擎名称"""
        ...

    @property
    @abstractmethod
    def version(self) -> str:
        """引擎版本"""
        ...

    @abstractmethod
    def calculate(self, input: EngineInput) -> Dict[str, Any]:
        """
        核心计算阶段
        - 纯算法计算，不涉及分析解读
        - 返回结构化计算结果
        """
        ...

    @abstractmethod
    def analyze(self, input: EngineInput, calculation: Dict) -> Dict[str, Any]:
        """
        分析阶段
        - 基于计算结果进行规则匹配和解读
        - 返回结构化分析结果
        """
        ...

    @abstractmethod
    def explain(self, input: EngineInput, calculation: Dict, analysis: Dict) -> str:
        """
        解释阶段
        - 生成人类可读的解释文本
        - 返回自然语言解释
        """
        ...

    def run(self, input: EngineInput) -> EngineOutput:
        """完整运行管道"""
        calc = self.calculate(input)
        analysis = self.analyze(input, calc)
        explanation = self.explain(input, calc, analysis)
        return EngineOutput(
            engine_name=self.name,
            input_hash=self._hash_input(input),
            timestamp=datetime.now(),
            calculation=calc,
            analysis=analysis,
            explanation=explanation,
            version=self.version,
            trace=self._build_trace(input, calc, analysis),
        )

    def _hash_input(self, input: EngineInput) -> str:
        import hashlib, json
        return hashlib.md5(json.dumps(input.params, sort_keys=True).encode()).hexdigest()[:12]

    def _build_trace(self, input, calc, analysis) -> Dict:
        return {
            "input": input.params,
            "calculation": calc,
            "analysis": analysis,
            "engine": self.name,
            "version": self.version,
        }
```

### 3.2 规则引擎（`rules/engine.py`）

```python
from dataclasses import dataclass, field
from typing import Any, Dict, List, Optional, Callable


@dataclass
class Rule:
    """单条规则"""
    id: str                         # 规则ID
    name: str                       # 规则名称
    category: str                   # 分类（如：bazi/格局、liuyao/用神）
    condition: str                  # 条件表达式（JSON Path）
    weight: float = 1.0             # 权重
    priority: int = 0               # 优先级（越高越优先）
    source: str = ""                # 来源（古籍出处）
    explanation: str = ""           # 规则解释
    action: Optional[Callable] = None  # 触发动作（可选）
    metadata: Dict = field(default_factory=dict)


@dataclass
class RuleResult:
    """规则匹配结果"""
    rule: Rule
    matched: bool
    value: Any = None
    confidence: float = 0.0


class RuleEngine:
    """规则引擎：加载、匹配、执行规则"""

    def __init__(self, rules: List[Rule] = None):
        self._rules: Dict[str, Rule] = {}
        self._by_category: Dict[str, List[Rule]] = {}
        if rules:
            for r in rules:
                self.register(r)

    def register(self, rule: Rule):
        self._rules[rule.id] = rule
        self._by_category.setdefault(rule.category, []).append(rule)

    def load_from_file(self, filepath: str):
        """从YAML/JSON文件加载规则"""
        import yaml, json
        with open(filepath, 'r', encoding='utf-8') as f:
            if filepath.endswith('.yaml'):
                data = yaml.safe_load(f)
            else:
                data = json.load(f)
        for r in data.get('rules', []):
            self.register(Rule(**r))

    def match(self, category: str, context: Dict) -> List[RuleResult]:
        """匹配指定分类下的所有规则"""
        results = []
        for rule in self._by_category.get(category, []):
            matched = self._evaluate(rule.condition, context)
            results.append(RuleResult(
                rule=rule,
                matched=matched,
                confidence=rule.weight if matched else 0.0,
            ))
        return results

    def evaluate_all(self, context: Dict, categories: List[str] = None) -> List[RuleResult]:
        """评估所有规则（或指定分类）"""
        cats = categories or list(self._by_category.keys())
        results = []
        for cat in cats:
            results.extend(self.match(cat, context))
        return results

    def _evaluate(self, condition: str, context: Dict) -> bool:
        """评估条件表达式"""
        # 支持：JSON Path取值、比较运算符、逻辑运算符
        # 示例: "day_master.wuxing == '木' AND month_zhi in ['寅','卯','辰']"
        import operator
        # 简化版条件解析，实际实现用 AST 或安全 eval
        import ast
        try:
            # SAFETY: 仅支持受限表达式
            return self._safe_eval(condition, context)
        except Exception:
            return False

    def _safe_eval(self, expr: str, context: Dict) -> bool:
        """安全的条件表达式求值"""
        # 实际实现：解析为AST，限制操作符，注入context
        # 此处为简化示意
        return eval(expr, {"__builtins__": {}}, context)
```

### 3.6 推理链追踪（`reasoning/trace.py`）

```python
@dataclass
class ReasoningStep:
    """推理链中的一步"""
    step_id: str
    step_type: str          # "calculation" | "rule_match" | "knowledge_lookup" | "fusion" | "explanation"
    input: Dict[str, Any]
    output: Dict[str, Any]
    rules_triggered: List[str] = field(default_factory=list)
    knowledge_used: List[str] = field(default_factory=list)
    timestamp: float = field(default_factory=time.time)


class ReasoningTrace:
    """完整推理链"""

    def __init__(self, trace_id: str):
        self.trace_id = trace_id
        self.steps: List[ReasoningStep] = []
        self.start_time = time.time()

    def add_step(self, step: ReasoningStep):
        self.steps.append(step)

    def to_dict(self) -> Dict:
        return {
            "trace_id": self.trace_id,
            "duration_ms": (time.time() - self.start_time) * 1000,
            "steps": [
                {
                    "id": s.step_id,
                    "type": s.step_type,
                    "input_summary": {k: str(v)[:100] for k, v in s.input.items()},
                    "output_summary": {k: str(v)[:100] for k, v in s.output.items()},
                    "rules_triggered": s.rules_triggered,
                    "knowledge_used": s.knowledge_used,
                }
                for s in self.steps
            ],
        }

    def to_text(self) -> str:
        """生成推理链文本"""
        lines = [f"【推理链】ID: {self.trace_id}"]
        for s in self.steps:
            lines.append(f"  [{s.step_type}] {s.step_id}")
            if s.rules_triggered:
                lines.append(f"    触发规则: {', '.join(s.rules_triggered)}")
            if s.knowledge_used:
                lines.append(f"    引用知识: {', '.join(s.knowledge_used)}")
        return "\n".join(lines)
```

## 五、关键技术决策

| 决策 | 选择 | 原因 |
|------|------|------|
| 后端统一算法 | Python后端为唯一算法源 | 消除前后端双实现，单点维护 |
| 前端角色 | 纯展示+交互 | 不再承担计算逻辑 |
| 规则存储 | YAML文件 | 人类可读、可编辑、版本可追踪 |
| 知识存储 | JSON文件 | 结构化查询、程序可消费 |
| 数据库 | SQLite（可选） | 无外部依赖，轻量级 |
| 测试框架 | pytest | 业界标准 |
| 配置管理 | YAML + 环境变量 | 分层配置 |
| 依赖注入 | FastAPI Depends | 框架原生支持 |
| 引擎版本 | 借鉴 engine.version 属性 | 保留向后兼容 |
